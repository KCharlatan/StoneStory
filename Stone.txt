func Cooldowns()
  onCD.Clear()
  cdLine=6
  for abItem:abItems
    cd=item.GetCooldown(abItem［0］)
    ? cd>0
      onCD.Add(abItem［0］)
      cdStr=time.FormatDigital(cd,true)
      cdStr=string.Sub(cdStr,2)
      ?cd<300
        cdStr=＂ ＂+string.Sub(cdStr,1)
      >`10,@cdLine@,@cdStr@ @abItem［1］@
      cdLine++

func Stopwatch()
  ?totaltime<loc.bestTime
    sw=loc.bestTime-totaltime
    swRGB=#00FF00
  :?totaltime<loc.averageTime
    sw=loc.averageTime-totaltime
    swRGB=#FFFF00
  :
    sw=totaltime-loc.averageTime
    swRGB=#FF0000
  sw=time.FormatDigital(sw,true)
  >`1,23,@swRGB@,@sw@

func StartFrame()
  FID=foe.id
  FS=foe.state
  FT=foe.time
  IL=item.left
  ILS=item.left.state
  IR=item.right
  IRS=item.right.state
  pdHead=#E0E0E0
  pdL=StoneState(ILS,lastL3)
  pdR=StoneState(IRS,lastR3)
  lastL3=(ILS=3)
  lastR3=(IRS=3)
  fdClr=FoeState(FS)
  
func StoneState(state,lastState)
  ? state=2
    rgb=#0000FF
  :?state=3&lastState
    rgb=#FF0000
  :?state>=4|state<=0
    rgb=#FF8000
  :
    rgb=pdHead
  return rgb

func FoeState(state)
  ? state=1
    rgb=#FFFF00
  ? state=2
    rgb=#FF0000
  :?state=3
    rgb=#FF8000
  :?state=4
    rgb=#00FF00
  :
    rgb=#FFFFFF
  return rgb
